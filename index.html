aluminium_window_frame.scad 
// ============================================================
// PARAMETRIC ALUMINIUM WINDOW FRAME
// Author  : Timothy Tiamapti
// Version : 1.0
// Description: Fully parametric window frame with configurable
//              dimensions, wall thickness, and corner style.
//              Inspired by real fabrication work (ESILON Construction).
// Usage: Adjust parameters below and render with F5/F6
// ============================================================

// ── PARAMETERS ─────────────────────────────────────────────
frame_width      = 200;   // Overall width in mm
frame_height     = 150;   // Overall height in mm
frame_depth      = 60;    // Extrusion depth in mm
wall_thickness   = 4;     // Wall thickness in mm
corner_radius    = 2;     // Corner radius (0 = sharp)
num_dividers     = 1;     // Number of horizontal dividers
divider_thickness = 3;   // Divider thickness in mm
fn               = 64;   // Smoothness for curves

// ── COMPUTED VALUES ─────────────────────────────────────────
inner_w = frame_width  - 2 * wall_thickness;
inner_h = frame_height - 2 * wall_thickness;

// ── MODULE: Rounded Rectangle Profile ───────────────────────
module rounded_rect(w, h, r) {
    if (r > 0) {
        minkowski() {
            square([w - 2*r, h - 2*r], center=true);
            circle(r, $fn=fn);
        }
    } else {
        square([w, h], center=true);
    }
}

// ── MODULE: Frame Outer Shell ────────────────────────────────
module frame_shell() {
    linear_extrude(height=frame_depth) {
        difference() {
            // Outer profile
            rounded_rect(frame_width, frame_height, corner_radius);
            // Inner hollow cutout
            rounded_rect(inner_w, inner_h, max(0, corner_radius-wall_thickness));
        }
    }
}

// ── MODULE: Horizontal Dividers ──────────────────────────────
module dividers() {
    spacing = frame_height / (num_dividers + 1);
    for (i = [1 : num_dividers]) {
        y_pos = -frame_height/2 + i * spacing;
        translate([0, y_pos, frame_depth/2])
            cube([inner_w, divider_thickness, frame_depth], center=true);
    }
}

// ── MAIN ASSEMBLY ────────────────────────────────────────────
module window_frame() {
    color("silver", 0.9)
    union() {
        frame_shell();
        dividers();
    }
}

// ── RENDER ───────────────────────────────────────────────────
window_frame();

// ── OPTIONAL: View cross-section (uncomment to use)
// projection(cut=true) translate([0,0,-1]) window_frame();

Project 02 — Architectural Model
Parametric Cowshed v2.0 — Full Agricultural Complex
A fully parametric agricultural complex for 10 cows. Features individual stalls, worker path, feeding & water troughs, ventilated 3/4 walls, entrance gates, pitched roof with structural trusses, aluminium window frames, concrete foundation slab, rainwater gutters, downpipe, and a cylindrical collection tank with inlet and outlet. Every dimension is parameter-driven.
modules for-loop difference() polygon() linear_extrude rotate() color() parametric foundation trusses gutters water tank

cowshed_timothy.scad 
// ============================================================
// PARAMETRIC COWSHED v2.0 — Timothy Tiamapti
// Features: 10 stalls, worker path, feeding & water troughs,
//   3/4 ventilated walls, entrance gates, pitched roof,
//   roof trusses, window frames, foundation slab,
//   gutters, rainwater collection tank
// ============================================================

// ── PARAMETERS ─────────────────────────────────────────────
stall_w=150; stall_d=250; stall_h=200; wall_t=10;
num_stalls=5; path_w=120; water_d=80;
found_t=15;   // foundation thickness
found_ext=20; // foundation overhang
tank_r=80;    // water tank radius
tank_h=200;   // water tank height
gutter_w=15; gutter_h=10; // gutter dimensions
win_w=60; win_h=50; // window size
shed_w = num_stalls*stall_w + 2*wall_t;
shed_d = 2*stall_d + path_w + 2*wall_t;

// ── FOUNDATION SLAB ─────────────────────────────────────────
module foundation() {
  color("lightgray")
  translate([-found_ext,-found_ext,-found_t])
    cube([shed_w+2*found_ext, shed_d+2*found_ext, found_t]);
}

// ── WINDOW FRAME MODULE ──────────────────────────────────────
module window_frame(w, h) {
  color("silver")
  difference() {
    cube([w, wall_t+2, h]);
    translate([3, -1, 3])
      cube([w-6, wall_t+4, h-6]);
  }
  // centre divider
  translate([w/2-1.5, -1, 3])
    cube([3, wall_t+4, h-6]);
}

// ── PLACE WINDOWS ON LONG WALLS ─────────────────────────────
module all_windows() {
  win_z = stall_h*0.35;
  // front wall windows — one per stall
  for(i=[0:num_stalls-1]) {
    translate([wall_t+i*stall_w+(stall_w-win_w)/2,
               -1, win_z])
      window_frame(win_w, win_h);
  }
  // back wall windows
  for(i=[0:num_stalls-1]) {
    translate([wall_t+i*stall_w+(stall_w-win_w)/2,
               shed_d-wall_t-1, win_z])
      window_frame(win_w, win_h);
  }
}

// ── ROOF TRUSS MODULE ────────────────────────────────────────
module truss() {
  color("saddlebrown")
  tw = shed_d; th = shed_d/4; tb = 8;
  union() {
    // left rafter
    rotate([0, 0, 0])
    linear_extrude(height=tb)
      polygon([[0,0],[tw/2,th],[tw/2-tb,th],[tb,0]]);
    // right rafter
    linear_extrude(height=tb)
      polygon([[tw,0],[tw/2,th],[tw/2+tb,th],[tw-tb,0]]);
    // bottom chord (tie beam)
    cube([tw, tb, tb]);
    // vertical king post
    translate([tw/2-tb/2, 0, 0])
      cube([tb, tb, th]);
    // diagonal struts
    linear_extrude(height=tb)
      polygon([[tw/4,0],[tw/2,th/2],[tw/2,th/2-tb],[tw/4+tb,0]]);
    linear_extrude(height=tb)
      polygon([[tw*3/4,0],[tw/2,th/2],[tw/2,th/2-tb],[tw*3/4-tb,0]]);
  }
}

// ── PLACE TRUSSES ALONG ROOF ─────────────────────────────────
module roof_trusses() {
  spacing = shed_w / (num_stalls+1);
  for(i=[1:num_stalls]) {
    translate([i*spacing-4, 0, stall_h+10])
      rotate([90,0,90]) truss();
  }
}

// ── GUTTERS ──────────────────────────────────────────────────
module gutter(length) {
  color("steelblue")
  difference() {
    cube([length, gutter_w, gutter_h]);
    translate([0, 2, 2])
      cube([length, gutter_w-4, gutter_h]);
  }
}

module all_gutters() {
  gutter_z = stall_h - 5;
  // front gutter
  translate([0, -gutter_w, gutter_z])
    gutter(shed_w);
  // back gutter
  translate([0, shed_d, gutter_z])
    gutter(shed_w);
  // downpipe to tank (front right corner)
  color("steelblue")
  translate([shed_w-gutter_w, -gutter_w, 0])
    cube([gutter_w, gutter_w, gutter_z]);
}

// ── RAINWATER TANK ───────────────────────────────────────────
module water_tank() {
  color("darkgreen", 0.85)
  translate([shed_w+30, 0, 0]) {
    // tank body
    difference() {
      cylinder(h=tank_h, r=tank_r, $fn=64);
      translate([0,0,5])
        cylinder(h=tank_h, r=tank_r-5, $fn=64);
    }
    // tank lid
    color("darkgreen")
    translate([0,0,tank_h])
      cylinder(h=8, r=tank_r+2, $fn=64);
    // inlet pipe from gutter
    color("steelblue")
    translate([-tank_r-30, 0, tank_h-20])
      rotate([0,90,0])
        cylinder(h=30, r=6, $fn=32);
    // outlet tap at base
    color("silver")
    translate([-tank_r-10, 0, 20])
      rotate([0,90,0])
        cylinder(h=12, r=4, $fn=32);
  }
}

module cowshed_v2() {
  foundation();
  shed_walls();
  left_stalls();   right_stalls();
  worker_path();
  water_trough();  feeding_trough();
  all_windows();
  roof();
  roof_trusses();
  all_gutters();
  water_tank();
}
